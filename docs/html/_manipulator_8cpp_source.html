69// вычисление позиции звена в глобальной системе координат
70// 1. roll не влияет на позицию (в нашем роботе для упрощения стержни вращаются так, что не влияют на другие стержни)
71// 2. координаты звена n: (x_n, y_n, z_n)[абс] = (x_{n-1}, y_{n-1}, z_{n-1})[абс] + (r_n*cos(yaw)*sin(pitch), r_n*sin(yaw)*sin(pitch), r_n*cos(pitch))
72// где [абс] - координаты в глобальной системе координат
73// 3. представим, что первое звено прикреплено к блоку управления, причем понятно, что pitch(id=1)<=pi/2, yaw(id=1)<=pi/2, см. картинку в readme;
74// в то время как pitch(id>1), yaw(id>1) - любые с условием, что звенья не сталкиваются друг с другом. Этот блок управления - ноль в глобальной системе координат.
75// Тогда по формуле выше (x_{1}, y_{1}, z_{1})[абс] = (r{n}*cos(yaw)sin(pitch), r{n}*sin(yaw)sin(pitch), r{n}*cos(pitch)).
76// 4. Далее с помощью for находим координаты всех звеньев вплоть до n-го
77std::tuple<double, double, double> Manipulator::calculatePosition(int id) const {
78 // если звена с таким id нет, бросаем исключение — вызывающий код (Movement) ловит это
79 if (!links_.count(id)) {
80 throw std::runtime_error("link with this id doesn't exist");
81 }
82
83 // соберем все звенья от базы (prev_id == 0) до нужного звена в правильном порядке
84 std::vector<int> chain;
85 int current_id = id;
86 while (current_id != 0) {
87 auto it = links_.find(current_id);
88 if (it == links_.end()) {
89 // ДЕМОНСТРАЦИЯ ОБРАБОТКИ ОШИБОК ИЗ БИБЛИОТЕКИ (Этап 4):
90 // Бросаем специальное исключение ManipulatorException вместо std::runtime_error.
91 // Это исключение будет поймано в c_api.cpp и обработано специальным образом,
92 // что позволит передать информацию об ошибке в графический интерфейс.
93 throw ManipulatorException("Нарушена цепочка звеньев до звена с ID=" + std::to_string(id) +
94 ". Звено с ID=" + std::to_string(current_id) + " не найдено.");
95 }
96 chain.push_back(current_id);
97 current_id = it->second->getPrevId();
98 }
99 std::reverse(chain.begin(), chain.end());
100
101 // начальные координаты и набранные углы
102 double x = 0.0, y = 0.0, z = 0.0;
103 double accum_pitch = 0.0, accum_yaw = 0.0, accum_roll = 0.0;
104
105 for (size_t idx = 0; idx < chain.size(); ++idx) {
106 int lid = chain[idx]; // lid == link id
107 MovableLink* link = links_.at(lid);
108
109 double r = link->getR();
110 auto dir = link->getDirection();
111 double pitch = dir.first;
112 double yaw = dir.second;
113 double roll = 0.0; // roll в нашей модели не влияет на угол
114
115 accum_pitch += pitch;
116 accum_yaw += yaw;
117 accum_roll += roll;
118
119 double dx = r * std::cos(accum_yaw) * std::sin(accum_pitch);
120 double dy = r * std::sin(accum_yaw) * std::sin(accum_pitch);
121 double dz = r * std::cos(accum_pitch);
122
123 x += dx;
124 y += dy;
125 z += dz;
126
127 // проверка столкновений
128 if (idx > 0) {
129 if (!checkCollision(chain, static_cast<int>(idx), x, y, z)) {
130 throw std::runtime_error("Collision detected for link " + std::to_string(lid));
131 }
132 }
133 }
134
135 return std::make_tuple(x, y, z);
136}

140// проверка столкновений: проверяем, что новое звено не пересекается
141// с предыдущими звеньями в цепочке
142bool Manipulator::checkCollision(const std::vector<int>& chain, size_t current_index,
143 double x, double y, double z) const {
144
145 // получим координаты текущего звена
146 double current_x = x;
147 double current_y = y;
148 double current_z = z;
149
150 // проверим столкновения с предыдущими звеньями в цепочке
151 for (size_t i = 0; i < current_index; ++i) {
152 // вычислим координаты предыдущего звена
153 double prev_x = 0.0, prev_y = 0.0, prev_z = 0.0;
154 for (size_t j = 0; j <= i; ++j) {
155 const MovableLink* link = links_.at(chain[j]);
156 double r = link->getR();
157 auto direction = link->getDirection();
158 double pitch = direction.first;
159 double yaw = direction.second;
160
161 double dx = r * cos(yaw) * sin(pitch);
162 double dy = r * sin(yaw) * sin(pitch);
163 double dz = r * cos(pitch);
164
165 prev_x += dx;
166 prev_y += dy;
167 prev_z += dz;
168 }
169
170 // проверим расстояние между звеньями
171 double distance = sqrt(pow(current_x - prev_x, 2) +
172 pow(current_y - prev_y, 2) +
173 pow(current_z - prev_z, 2));
174
175 // если расстояние слишком мало, считаем это столкновением
176 if (distance < 0.1) { // минимальное расстояние между звеньями (по факту проверяем лишь то, что звенья не сомкнулись в цикл)
177 return false;
178 }
179 }
180
181 return true;
182}
