123int rm_calculate_position(RM_Manipulator manip, int id, double* out_x, double* out_y, double* out_z) {
124 // Вычисляет позицию звена и записывает в переданные указатели.
125 // Возвращает 1 при успехе, 0 при общей ошибке, -1 при ManipulatorException.
126 if (!manip || !out_x || !out_y || !out_z) return 0;
127 try {
128 Manipulator* m = reinterpret_cast<Manipulator*>(manip);
129 auto pos = m->calculatePosition(id);
130 *out_x = std::get<0>(pos);
131 *out_y = std::get<1>(pos);
132 *out_z = std::get<2>(pos);
133 return 1;
134 } catch (const ManipulatorException& e) {
135 // ДЕМОНСТРАЦИЯ ОБРАБОТКИ ОШИБОК ИЗ БИБЛИОТЕКИ (Этап 4):
136 // Специальное исключение ManipulatorException из библиотеки ловится здесь конкретно,
137 // а не через catch(...), что позволяет передать информацию об ошибке в интерфейс.
138 // Возвращаем код -1, который будет обработан в Python UI для показа пользователю
139 // понятного сообщения об ошибке.
140 std::cerr << "ManipulatorException: " << e.what() << std::endl;
141 return -1;
142 } catch (...) {
143 // Все остальные исключения - общая ошибка
144 return 0;
145 }
146}
